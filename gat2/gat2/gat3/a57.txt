        document.getElementById('file').addEventListener('change', function(e) {
            const file = e.target.files[0];
            if(file) {
                document.getElementById('fileInfo').innerHTML = 
                    `文件：${file.name} (${(file.size/1024).toFixed(1)}KB)`;

        function toggleMode(mode) {
            document.getElementById('uploadSection').style.display = mode === 'upload' ? 'block' : 'none';
            document.getElementById('txtSection').style.display = mode === 'txt' ? 'block' : 'none';
        }

        function readFile() {
            const filename = document.getElementById('filename').value;
            const password = document.getElementsByName('password')[0].value;
            const xhr = new XMLHttpRequest();
            xhr.open('GET', 'read.php?filename=' + encodeURIComponent(filename) + '&password=' + encodeURIComponent(password), true);
            
            document.getElementById('loadingMsg').style.display = 'inline';
            document.getElementById('linkAnchor').style.display = 'none';

            xhr.onload = function() {
                document.getElementById('loadingMsg').style.display = 'none';
                if (xhr.status === 200) {
                    const response = JSON.parse(xhr.responseText);
                    if (response.error) {
                        alert(response.error);
                    } else {
                        document.getElementsByName('content')[0].value = response.content;
                        document.getElementById('linkAnchor').href = response.fileLink;
                        document.getElementById('linkAnchor').textContent = response.fileLink;
                    }
                } else {
                    alert('读取失败: ' + xhr.statusText);
                }
                document.getElementById('linkAnchor').style.display = 'inline';
            };

            xhr.onerror = function() {
                document.getElementById('loadingMsg').style.display = 'none';
                alert('网络请求失败');
            };

            xhr.send();
        }
    </script>
